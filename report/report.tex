\subsection{Estratégia escolhida: Teste de Classe Não-Modal}
\label{sec:order-strategy}

A classe \classname{Order} não possui uma máquina de estados (não há modos que alterem o comportamento dos métodos); o seu comportamento ao nível de classe é determinado por um conjunto de \emph{invariantes} sobre as suas variáveis de instância. A estratégia apropriada para testar uma classe deste tipo é o \textbf{Teste de Classe Não-Modal}, que consiste em: (i) identificar os domínios das variáveis; (ii) determinar os pontos \emph{On}/\emph{Off} de cada fronteira de invariante; e (iii) construir uma \emph{matriz de domínio} com vetores de teste, em que cada caso de teste exercita uma fronteira de cada vez, mantendo as restantes variáveis em valores típicos.

\section{Concretização em TestNG}
\label{sec:testng}

Foram seleccionados $4$ casos de sucesso e $4$ casos de insucesso da bateria de testes da classe \classname{Order} (Secção~\ref{sec:order-class}).

\begin{lstlisting}[style=java,caption={Esqueleto da classe de testes TestNG para \classname{Order}.},label=lst:testng]
package supereats.tests;

import org.testng.annotations.*;
import static org.testng.Assert.*;

import supereats.core.*;

public class OrderTest {

    private static final double EPS = 1e-9;
 
    private Supplier s1;   // fornecedor de referencia
    private Supplier s2;   // fornecedor distinto
    private Client   client;
 
    // Itens do fornecedor s1 (weight em gramas, price inteiro)
    private Item i1;       // preco 10
    private Item i2;       // preco 5
    private Item i50;      // preco 50  (para casos de totalQty)
    // Item do fornecedor distinto s2
    private Item iOther;

    @BeforeMethod
    public void setUp() {
        s1     = new Supplier("Loja A");
        s2     = new Supplier("Loja B");
        client = newClient();              // cliente valido qualquer
 
        i1     = new Item(100, 10, s1, "Item 1");
        i2     = new Item(200,  5, s1, "Item 2");
        i50    = new Item(150, 50, s1, "Item 50");
        iOther = new Item(100, 10, s2, "Item de outro fornecedor");
    }
 
    // Auxiliar: assume-se a existencia de um construtor de Client.
    private Client newClient() {
        return new Client(/* ... dados validos ... */);
    }
    // ---------- Casos de SUCESSO ----------

    // TC1: construcao valida - todos os invariantes satisfeitos.
    @Test
    public void tc1_validConstruction() {
        Order o = new Order(client, "Rua A", 5000);
        assertEquals(o.totalQuantity(), 0);
        assertEquals(o.cost(), 0.0, EPS);
        assertTrue(o.itens().isEmpty());
    }

    @Test
    public void T9_addItem_emptyOrder_validQuantity() {
        Order o = new Order(client, "Rua A", 5000);
        o.addItem(i1, 1);
        assertTrue(o.contains(i1));
        assertEquals(o.quantity(i1), 1);
        assertEquals(o.cost(), 10.0, EPS);
    }

    
    // TC13: fronteira totalQty <= 20 + floor(cost/50) (On).
    // Usa-se preco unitario 1 para isolar a fronteira de totalQty sem
    // violar o limite de custo (I3). Com cost = 24 * 1 = 24, o limite e'
    // 20 + floor(24/50) = 20 + 0 = 20... pelo que e' preciso elevar o
    // custo o suficiente para que o limite atinja 24:
    //   limite = 24  <=>  floor(cost/50) = 4  <=>  cost em [200, 250).
    // Escolhe-se cost = 240 com itens de preco 10 (24 unidades * 10 = 240),
    // mantendo cost <= 1000.
    @Test
    public void tc13_totalQuantityAtDynamicLimit() {
        Order o = new Order(client, "Rua A", 5000);
        // 3 itens de preco 10, quantidades 10 + 10 + 4 = 24 unidades
        Item a = new Item(150, 10, s1, "A");
        Item b = new Item(150, 10, s1, "B");
        Item c = new Item(150, 10, s1, "C");
        o.addItem(a, 10);
        o.addItem(b, 10);
        o.addItem(c, 4);
        assertEquals(o.totalQuantity(), 24);   // = 20 + floor(240/50) = 24
        assertEquals(o.cost(), 240.0, EPS);
    }
 
    // TC19: - adicionar item do MESMO fornecedor é aceite.
    @Test
    public void tc19_sameSupplier_isAccepted() {
        Order o = new Order(client, "Rua A", 5000);
        o.addItem(i1, 2);
        o.addItem(i2, 3);                      // i2 tambem do fornecedor s1
        assertTrue(o.contains(i1));
        assertTrue(o.contains(i2));
        assertEquals(o.totalQuantity(), 5);
        assertEquals(o.cost(), 2*10 + 3*5, EPS);
    }

    // ---------- Casos de INSUCESSO ----------

    // TC2: fronteira addrLength >= 1 (Off) - morada vazia.
    @Test(expectedExceptions = InvalidOperationException.class)
    public void tc2_emptyAddress_throws() {
        new Order(client, "", 5000);
    }
 
    // TC6: cliente null.
    @Test(expectedExceptions = InvalidOperationException.class)
    public void tc6_nullClient_throws() {
        new Order(null, "Rua A", 5000);
    }
 
    // TC10: fronteira quantity <= 10 (Off) - quantidade 11.
    // Verifica tambem que o estado nao e' alterado (I6).
    @Test
    public void tc10_quantityAboveMax_throwsAndKeepsState() {
        Order o = new Order(client, "Rua A", 5000);
        try {
            o.addItem(i1, 11);
            fail("Esperava InvalidOperationException");
        } catch (InvalidOperationException e) {
            assertFalse(o.contains(i1));       // estado inalterado
            assertEquals(o.totalQuantity(), 0);
        }
    }
 
    // TC20: adicionar item de fornecedor DISTINTO.
    @Test(expectedExceptions = InvalidOperationException.class)
    public void tc20_differentSupplier_throws() {
        Order o = new Order(client, "Rua A", 5000);
        o.addItem(i1, 1);                      // fornecedor s1
        o.addItem(iOther, 1);                  // fornecedor s2 -> excecao
    }
}
